\begin{question}{r203/2023/cours/rsync}
  Quelle peut être l'utilité de l'outil {\em rsync} ?
  \begin{reponses}[2]
    \bonne{synchroniser les contenus de fichiers/répertoires}
    \mauvaise{chiffrer le contenu d'un fichier}
    \mauvaise{synchroniser les horloges d'un réseau}
    \mauvaise{se connecter sur un hôte distant (sans chiffrement)}
    \lastchoices
    \mauvaise{ARNC}
  \end{reponses}
\end{question}
