\begin{question}{r203/2023/ssh/cmd-ssh1}
  Alice est connectée sur sa machine en tant qu'utilisateur
  alice. Quelle commande peut-elle exécuter pour ouvrir une session
  SSH en tant qu'utilisateur {\em alice} sur la machine de Bob
  (1.2.3.4) ?
  \begin{reponses}[3]
    \bonne{{\tt ssh 1.2.3.4}}
    \mauvaise{{\tt ssh alice 1.2.3.4}}
    \mauvaise{{\tt ssh 1.2.3.4:alice}}
    \mauvaise{{\tt ssh @1.2.3.4}}
    \mauvaise{{\tt ssh 1.2.3.4@alice}}
    \mauvaise{{\tt /etc/init.d/sshd 1.2.3.4}}
    \lastchoices
    \mauvaise{ARNC}
  \end{reponses}
\end{question}
