\begin{question}{r203/2023/ssh/message}
  Vous voulez envoyer un message chiffré à Anna en utilisant un
  chiffrement asymétrique. Avec quelle(s) clé(s) allez vous chiffrer
  votre message ?
  \begin{reponses}[2]
    \bonne{la clé publique d'Anna}
    \mauvaise{votre clé publique}
    \mauvaise{votre clé privée + la clé publique d'Anna}
    \mauvaise{la clé privée d'Anna}
  \end{reponses}
\end{question}
