\begin{question}{r203/2023/ssh/cmd-ssh3}
  Alice est connectée sur sa machine en tant qu'utilisateur
  alice. Quelle commande peut-elle exécuter pour afficher le contenu
  du répertoire \texttt{/tmp/} se trouvant sur la machine de Bob
  (1.2.3.4) ?
  \begin{reponses}[3]
    \bonne{{\tt ssh 1.2.3.4 ls /tmp/}}
    \mauvaise{{\tt scp 1.2.3.4 ls /tmp/}}
    \mauvaise{{\tt scp 1.2.3.4:/tmp/}}
    \mauvaise{{\tt ssh 1.2.3.4 | ls /tmp}}
    \mauvaise{{\tt ssh 1.2.3.4:/tmp/}}
    \mauvaise{{\tt scp 1.2.3.4:/tmp .}}
    \lastchoices
    \mauvaise{ARNC}
  \end{reponses}
\end{question}
