%webamc RND=1
\begin{question}{mcq-random/fp/00}
  %webamc BEGIN
  \FPeval\VARa{trunc(1+random*8,0)}
  \FPeval\VARb{trunc(4+random*5,0)}
  \FPeval\VARcorrect{clip(VARa+VARb)}
  \FPeval\VARwronga{clip(VARa+VARb-1)}
  \FPeval\VARwrongb{clip(VARa+VARb+1)}
  \FPeval\VARwrongc{clip(VARa-VARb)}
  %webamc END
  What is the sum of \VARa{} and \VARb{}~?
  \begin{choices}
    \correctchoice{\VARcorrect}
    \wrongchoice{\VARwronga}
    \wrongchoice{\VARwrongb}
    \wrongchoice{\VARwrongc}
  \end{choices}
\end{question}
