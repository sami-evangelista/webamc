\begin{question}{r316/2024/cours/sip-ua-field}
  Dans un message SIP/Invite, quel champ d'en-tête permet d'identifier
  le logiciel ou le type d'équipement de téléphonie
  initiant l'appel~?
  \begin{reponses}
    \bonne{\tt User-Agent}
    \mauvaise{\tt Agent-Type}
    \mauvaise{\tt User-Type}
    \mauvaise{\tt Client-Agent}
    \mauvaise{\tt Client-Type}
  \end{reponses}
\end{question}
