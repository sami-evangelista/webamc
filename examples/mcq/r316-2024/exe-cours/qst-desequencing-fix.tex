\begin{question}{r316/2024/cours/desequencing-fix}
  Un UA peut-il corriger une erreur de déséquencement de paquets
  (i.e., paquets reçus dans le désorde)~?
  \begin{reponses}[2]
    \bonne{oui, grâce à son tampon en réception}
    \mauvaise{oui, grâce à son tampon en émission}
    \mauvaise{oui, avec certains codecs}
    \mauvaise{non, c'est impossible}
  \end{reponses}
\end{question}
