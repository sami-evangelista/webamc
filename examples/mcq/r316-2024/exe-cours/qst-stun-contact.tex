\begin{question}{r316/2024/cours/stun-contact}
  Un UA est configuré pour utilisé un serveur STUN. Quand ce serveur
  STUN sera-t-il contacté par l'UA~?
  \begin{reponses}[2]
    \bonne{à chaque appel, avant l'établissement de la session}
    \mauvaise{à l'enregistrement de l'UA auprès de son registrar}
    \mauvaise{à chaque appel, après l'établissement de la session}
    \mauvaise{au premier appel, après l'établissement de la session}
    \mauvaise{au premier appel, avant l'établissement de la session}
  \end{reponses}
\end{question}
