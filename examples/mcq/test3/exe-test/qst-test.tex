\begin{question}{addition}
  Quelle est la somme de \QIa{} et \QIb{} ?
  \begin{reponses}
    \bonne{\QIsomme}
    \mauvaise{\QIerrA}
    \mauvaise{\QIerrB}
  \end{reponses}
\end{question}

\begin{question}{multiplication}
  Combien font \QIIa{} multiplié par \QIIb{} ?
  \begin{reponses}
    \bonne{\QIIprod}
    \mauvaise{\QIIerrA}
    \mauvaise{\QIIerrB}
  \end{reponses}
\end{question}

\begin{question}{soustraction}
  Calculez la différence : \QIIIa{} $-$ \QIIIb{}.
  \begin{reponses}
    \bonne{\QIIIdiff}
    \mauvaise{\QIIIerrA}
    \mauvaise{\QIIIerrB}
  \end{reponses}
\end{question}
