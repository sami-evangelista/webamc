\begin{question}{linux/bases/ls-tous}
  Quelle option de la commande {\tt ls} permet d'afficher les fichiers
  cachés (i.e., ceux dont le nom commence par un {\tt .})~?
  \begin{reponses}
    \bonne{\texttt{-a}}
    \mauvaise{\texttt{--t}}
    \mauvaise{\texttt{--r}}
    \mauvaise{\texttt{--i}}
    \mauvaise{\texttt{--l}}
    \mauvaise{\texttt{--d}}
    \mauvaise{\texttt{--u}}
    \lastchoices
    \mauvaise{Aucune des réponses précédentes ne convient.}
  \end{reponses}
\end{question}
