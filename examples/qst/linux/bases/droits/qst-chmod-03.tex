\begin{question}{linux/bases/droits/chmod-03}
  On suppose que les droits associés au fichier {\tt fic} sont {\tt
    rwxr-xr--}. On exécute la commande {\tt chmod g+w fic}. Quels
  seront alors les droits associés à {\tt fic}~?
  \begin{reponses}
    \bonne{\texttt{rwxrwxr--}}
    \mauvaise{\texttt{-w--w--w-}}
    \mauvaise{\texttt{r-xrwxr--}}
    \mauvaise{\texttt{r-xrwxrw-}}
    \mauvaise{\texttt{rw-rw-rw-}}
    \mauvaise{\texttt{rwxr-xr--}}
    \mauvaise{\texttt{rwxrwxrw-}}
    \lastchoices
    \mauvaise{Aucune des réponses précédentes ne convient.}
  \end{reponses}
\end{question}
