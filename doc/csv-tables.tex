\subsection{Table \texttt{grp}}
\begin{itemize}[itemsep=0pt,parsep=0pt,topsep=0pt,partopsep=0pt]
\item Description: This table defines user groups.
\item Columns:
\begin{itemize}[itemsep=0pt,parsep=0pt,topsep=0pt,partopsep=0pt]
\item\texttt{grp\_name} --- unique group name
\item\texttt{grp\_parent} --- parent group name (possibly empty, references {\tt
           grp.grp\_name})
\end{itemize}
\end{itemize}
\subsection{Table \texttt{tag}}
\begin{itemize}[itemsep=0pt,parsep=0pt,topsep=0pt,partopsep=0pt]
\item Description: This table defines tags that can be associated to MCQ
           items (e.g., questions).
\item Columns:
\begin{itemize}[itemsep=0pt,parsep=0pt,topsep=0pt,partopsep=0pt]
\item\texttt{tag\_name} --- unique tag name
\item\texttt{tag\_desc} --- tag description (possibly empty)
\item\texttt{tag\_color} --- tag color of the form \verb+#RRGGBB+ (possibly empty)
\end{itemize}
\end{itemize}
\subsection{Table \texttt{tbl}}
\begin{itemize}[itemsep=0pt,parsep=0pt,topsep=0pt,partopsep=0pt]
\item Description: This table contains the list of names of tables
           which can be administrated through the web interface, i.e.,
           for which CSV file submission is allowed.
\item Columns:
\begin{itemize}[itemsep=0pt,parsep=0pt,topsep=0pt,partopsep=0pt]
\item\texttt{tbl\_name} --- unique table name
\end{itemize}
\end{itemize}
\subsection{Table \texttt{local\_auth}}
\begin{itemize}[itemsep=0pt,parsep=0pt,topsep=0pt,partopsep=0pt]
\item Description: This table controls local system
           authentication. Each row gives access to one user. The
           login used by the user to authenticate is her/his code. An
           empty password prevents the user from logging meaning that
           he/she has to first reset his/her password. If not empty,
           the password is given plain.
\item Columns:
\begin{itemize}[itemsep=0pt,parsep=0pt,topsep=0pt,partopsep=0pt]
\item\texttt{loc\_enabled} --- 1 if local authentication is enabled for the
               user, 0 otherwise
\item\texttt{loc\_login} --- login of the user
\item\texttt{loc\_password} --- plain-text password (possibly empty)
\item\texttt{loc\_usr} --- user which is granted access (references {\tt
               usr.usr\_code})
\end{itemize}
\end{itemize}
\subsection{Table \texttt{cas\_auth}}
\begin{itemize}[itemsep=0pt,parsep=0pt,topsep=0pt,partopsep=0pt]
\item Description: This table controls CAS authentication. Each row
           gives access to one user.
\item Columns:
\begin{itemize}[itemsep=0pt,parsep=0pt,topsep=0pt,partopsep=0pt]
\item\texttt{cas\_enabled} --- 1 if CAS authentication enabled for the user, 0 otherwise
\item\texttt{cas\_login} --- login of the user on the CAS server
\item\texttt{cas\_usr} --- user which is granted access (references {\tt
               usr.usr\_code})
\end{itemize}
\end{itemize}
\subsection{Table \texttt{attr}}
\begin{itemize}[itemsep=0pt,parsep=0pt,topsep=0pt,partopsep=0pt]
\item Description: This table lists attributes that may be associated to
           users (e.g., student ID).
\item Columns:
\begin{itemize}[itemsep=0pt,parsep=0pt,topsep=0pt,partopsep=0pt]
\item\texttt{atr\_code} --- unique attribute code
\item\texttt{atr\_desc} --- a description of the attribute
\end{itemize}
\end{itemize}
\subsection{Table \texttt{usr}}
\begin{itemize}[itemsep=0pt,parsep=0pt,topsep=0pt,partopsep=0pt]
\item Description: This table defines users.
\item Columns:
\begin{itemize}[itemsep=0pt,parsep=0pt,topsep=0pt,partopsep=0pt]
\item\texttt{usr\_code} --- unique user code
\item\texttt{usr\_eaddr} --- unique electronic address
\item\texttt{usr\_fst\_name} --- first name
\item\texttt{usr\_name} --- last name
\end{itemize}
\end{itemize}
